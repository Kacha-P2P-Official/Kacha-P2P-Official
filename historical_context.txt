  const { ref, inView } = useInView(0.3);
  const { prefix, numeric, suffix } = parseStatValue(value);
  const animated = useCountUp({ target: numeric, duration: 1600, enabled: inView, originalStr: value });

  const isAnimatable = numeric > 0;

  return (
    // Use inView state directly for opacity so it works even without Tailwind intersect plugin
    <div ref={ref} className={`px-4 md:px-8 py-4 transition-opacity duration-700 ${inView ? 'opacity-100' : 'opacity-0'}`}>
      <div className="text-2xl md:text-3xl font-heading font-bold gold-text">

      </div>
      <div className="text-xs text-muted-foreground font-sans mt-1 uppercase tracking-wider">{label}</div>
    </div>
  );
